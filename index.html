async function loadCurrency() {
var result = await sb.from("currencies").select("*").eq("is_default", true).limit(1).single();
if (!result.error && result.data) activeCurrency = { symbol: result.data.symbol, rate: result.data.rate };
}

async function loadProducts() {
var result = await sb.from("products").select("*").order("id");
if (result.error) {
document.getElementById("productsGrid").innerHTML = "<p>Error loading products</p>";
return;
}
products = result.data;
renderFilters();
renderProducts();
}

function renderFilters() {
var cats = [];
for (var i = 0; i < products.length; i++) {
if (products[i].category && cats.indexOf(products[i].category) === -1) cats.push(products[i].category);
}
var el = document.getElementById("filters");
var allLabel = translations.allCategory[currentLang];
var html = '<button class="filter-btn' + (activeCategory === 'all' ? ' active' : '') + '" onclick="setCategory(\'all\')">' + allLabel + '</button>';
for (var j = 0; j < cats.length; j++) {
var c = cats[j];
html += '<button class="filter-btn' + (c === activeCategory ? ' active' : '') + '" onclick="setCategory(\'' + c + '\')">' + c + '</button>';
}
el.innerHTML = html;
}

function setCategory(c) { activeCategory = c; renderFilters(); renderProducts(); }

function renderProducts() {
var list = [];
for (var i = 0; i < products.length; i++) {
if (activeCategory === "all" || products[i].category === activeCategory) list.push(products[i]);
}
var query = document.getElementById("searchInput").value.trim().toLowerCase();
if (query) {
var filtered = [];
for (var j = 0; j < list.length; j++) {
var p = list[j];
var haystack = (p.name + " " + (p.description || "") + " " + (p.category || "")).toLowerCase();
if (haystack.indexOf(query) !== -1) filtered.push(p);
}
list = filtered;
}
var grid = document.getElementById("productsGrid");
if (list.length === 0) { grid.innerHTML = "<p>" + translations.noProducts[currentLang] + "</p>"; return; }
var html = "";
for (var k = 0; k < list.length; k++) {
var p = list[k];
var imgHtml = p.image_url ? '<img src="' + p.image_url + '" style="width:100%;height:100%;object-fit:contain;">' : '<i class="fa-solid fa-box"></i>';
var stockHtml = (p.in_stock === false || p.stock === 0) ? '<span class="out-badge">' + translations.outOfStock[currentLang] + '</span>' : '<button class="add-btn" onclick="addToCart(' + p.id + ')">' + translations.addToCart[currentLang] + '</button>';
html += '<div class="card"><div class="card-img" onclick="openProductDetail(' + p.id + ')">' + imgHtml + '</div><div class="card-body"><div class="card-cat">' + (p.category || "") + '</div><div class="card-title" onclick="openProductDetail(' + p.id + ')">' + p.name + '</div><div class="card-desc">' + (p.description || "") + '</div><div class="card-footer"><div class="card-price">' + formatPrice(p.price) + '</div>' + stockHtml + '</div></div></div>';
}
grid.innerHTML = html;
}

function openProductDetail(id) {
var p = null;
for (var i = 0; i < products.length; i++) {
if (products[i].id === id) { p = products[i]; break; }
}
if (!p) return;
document.getElementById("detailImg").src = p.image_url || "";
document.getElementById("detailImg").style.display = p.image_url ? "block" : "none";
document.getElementById("detailName").textContent = p.name;
document.getElementById("detailCat").textContent = p.category || "";
document.getElementById("detailDesc").textContent = p.description || "";
document.getElementById("detailPrice").textContent = formatPrice(p.price);
var addBtn = document.getElementById("detailAddBtn");
var outBadge = document.getElementById("detailOutBadge");
outBadge.textContent = translations.outOfStock[currentLang];
addBtn.textContent = translations.addToCart[currentLang];
if (p.in_stock === false || p.stock === 0) {
addBtn.style.display = "none";
outBadge.style.display = "inline";
} else {
addBtn.style.display = "inline-flex";
outBadge.style.display = "none";

document.getElementById("waAskBtn").querySelector("span").textContent = translations.askOnWhatsapp[currentLang];
document.getElementById("waAskBtn").href = "https://wa.me/2207472734?text=" + encodeURIComponent((currentLang === 'ar' ? 'بدي أعرف أكتر عن: ' : "I'd like to know more about: ") + p.name);

document.getElementById("detailOverlay").classList.add("show");
}

}

function addToCart(id) { cart[id] = (cart[id] || 0) + 1; updateCartUI(); }
function changeQty(id, delta) { cart[id] = (cart[id] || 0) + delta; if (cart[id] <= 0) delete cart[id]; updateCartUI(); }
function removeFromCart(id) { delete cart[id]; updateCartUI(); }

function updateCartUI() {
var count = 0;
for (var key in cart) { if (cart.hasOwnProperty(key)) count += cart[key]; }
document.getElementById("cartCount").textContent = count;
var body = document.getElementById("cartBody");
var ids = Object.keys(cart);
if (ids.length === 0) {
body.innerHTML = '<div class="empty-cart">' + translations.emptyCart[currentLang] + '</div>';
} else {
var html = "";
for (var i = 0; i < ids.length; i++) {
var id = parseInt(ids[i]);
var p = null;
for (var j = 0; j < products.length; j++) {
if (products[j].id === id) { p = products[j]; break; }
}
if (!p) continue;
html += '<div class="cart-item"><div class="cart-item-name">' + p.name + '<br><span style="color:#5b6478;font-size:12.5px;">' + formatPrice(p.price) + '</span></div><div class="qty-controls"><button onclick="changeQty(' + id + ',-1)">-</button><span>' + cart[id] + '</span><button onclick="changeQty(' + id + ',1)">+</button></div><i class="fa-solid fa-trash remove-x" onclick="removeFromCart(' + id + ')"></i></div>';
}
body.innerHTML = html;
}
var total = 0;
for (var k = 0; k < ids.length; k++) {
var id2 = parseInt(ids[k]);
for (var m = 0; m < products.length; m++) {
if (products[m].id === id2) { total += products[m].price * cart[id2]; break; }
}
}
document.getElementById("cartTotal").textContent = formatPrice(total);
var cartSummary = "";
for (var n = 0; n < ids.length; n++) {
var id3 = parseInt(ids[n]);
for (var o = 0; o < products.length; o++) {
if (products[o].id === id3) {
if (cartSummary) cartSummary += ", ";
cartSummary += products[o].name + " x" + cart[id3];
break;
}
}
}
var waMsg = (currentLang === 'ar' ? 'مرحباً، بدي أستفسر عن هالمنتجات: ' : "Hi, I'd like to ask about these items: ") + cartSummary;
document.getElementById("waCartBtn").href = "https://wa.me/2207472734?text=" + encodeURIComponent(waMsg);
}

function copyWaveNumber() {
var num = STORE_WAVE_NUMBER;

navigator.clipboard.writeText(num).then(function() {
alert(translations.waveCopied[currentLang]);

function fallbackCopy(text) {
var textarea = document.createElement("textarea");
textarea.value = text;
document.body.appendChild(textarea);
textarea.select();
try { document.execCommand("copy"); alert(translations.waveCopied[currentLang]); } catch (e) {}
document.body.removeChild(textarea);
}

}
closeWaveModal();
if (pendingWaveOrder) {
pendingWaveOrder.waveRef = ref;

}

async function submitOrder() {
var name = document.getElementById("custName").value.trim();
var phone = document.getElementById("custPhone").value.trim();
var address = document.getElementById("custAddress").value.trim();
var payment = document.querySelector('input[name="pay"]:checked').value;
if (!name || !phone || !address) { alert(translations.fillAllFields[currentLang]); return; }
var itemsText = "";
var total = 0;
var ids = Object.keys(cart);
for (var i = 0; i < ids.length; i++) {
var id = parseInt(ids[i]);
for (var j = 0; j < products.length; j++) {
if (products[j].id === id) {
if (itemsText) itemsText += ", ";
itemsText += products[j].name + " x" + cart[id];
total += products[j].price * cart[id];
break;
}
}
}
if (payment === "Wave") {
var orderData = {
name: name, phone: phone, address: address, paymentMethod: payment,
location: customerLocation || document.getElementById("custLocationLink").value.trim(),
itemsText: itemsText, total: total, waveRef: null
};
openWaveModal(orderData);
return;
}
await submitOrderFinal({
name: name, phone: phone, address: address, paymentMethod: payment,
location: customerLocation || document.getElementById("custLocationLink").value.trim(),
itemsText: itemsText, total: total, waveRef: null
});
}

async function submitOrderFinal(orderData) {
var name = orderData.name;
var phone = orderData.phone;
var address = orderData.address;
var location = orderData.location || "";
var itemsText = orderData.itemsText;
var total = orderData.total;
var paymentMethod = orderData.paymentMethod;
var waveRef = orderData.waveRef || null;
var result = await sb.from("orders").insert([{
customer_name: name, phone: phone, address: address,
location: location, items: itemsText, total_price: total,
payment_method: paymentMethod,
status: paymentMethod === "Wave" ? "بانتظار تأكيد الدفع" : "جديد",
wave_reference: waveRef
}]).select().single();
if (result.error) { alert(translations.orderError[currentLang]); return; }
var orderId = result.data.id;
var itemRows = [];
var ids = Object.keys(cart);
for (var i = 0; i < ids.length; i++) {
var id = parseInt(ids[i]);
for (var j = 0; j < products.length; j++) {
if (products[j].id === id) {
itemRows.push({
order_id: orderId,
product_name: products[j].name,
qty: cart[id],
unit_price: products[j].price,
line_total: products[j].price * cart[id],
description: products[j].description || "",
unit: "pcs"
});
break;
}
}
}
try { await sb.from("order_items").insert(itemRows); } catch(e) {}
try {
var paymentNote = paymentMethod === "Wave" ? "(Wave Ref: " + (waveRef || 'لم يتم إدخال مرجع') + ")" : "";
await fetch(FORMSPREE_URL, {
method: "POST",
headers: { "Content-Type": "application/json", "Accept": "application/json" },
body: JSON.stringify({
subject: "طلب جديد من المتجر - New Store Order",
name: name, phone: phone, address: address, items: itemsText,
total: "$" + total, payment_method: paymentMethod + " " + paymentNote,
wave_reference: waveRef || "N/A"
})
});
} catch(e) {}
alert(translations.orderSuccess[currentLang]);
cart = {};
customerLocation = "";
var locBtn = document.getElementById("locationBtn");
locBtn.classList.remove("got-location");
locBtn.innerHTML = '<i class="fa-solid fa-location-dot"></i> <span data-i18n="shareLocation">' + translations.shareLocation[currentLang] + '</span>';
updateCartUI();
closeCart();
}
